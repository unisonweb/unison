``` unison
Log.lineSwapped : Nat -> Text -> Text -> Text
Log.lineSwapped code level message =
  use Text ++
  "[" ++ level ++ " " ++ Nat.toText code ++ "] " ++ message

swap level code = @rewrite term Log.line level code ==> Log.lineSwapped code level
```
