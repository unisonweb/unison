-- | Github request data type.

-- * @mt@ describes the media type, i.e. how the response should be interpreted.
-- * @a@ is the result type

    Query        :: Paths -> QueryString -> GenRequest mt rw a

        -> Paths                   -- ^ path
        -> LBS.ByteString          -- ^ body
        -> GenRequest mt 'RW a
