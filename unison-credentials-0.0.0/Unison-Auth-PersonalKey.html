Sign arbitrary bytes using a personal private key

>>> key <- generatePersonalKey
>>> let msg = "Hello, world!"
>>> signature <- fromRight (error "failed to sign") <$> signWithPersonalKey key msg
>>> verifyWithPersonalKey (publicKey key) msg signature
