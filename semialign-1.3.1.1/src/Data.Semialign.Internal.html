-- non-uniform shapes.
--
-- Minimal definition: either 'zip' or 'zipWith'.
--
-- /Idempotency/
--
-- @
-- join zip   ≡ fmap (join (,))

--
-- /Commutativity/
--
-- @
-- zip x y ≡ swap \<$> zip y x

-- /Associativity/
--
-- @
-- zip x (zip y z) ≡ assoc \<$> zip (zip x y) z

--     toThis (This a)    = This a

--

-- zipWith f a b ≡ f \<$> zip a b

-- fmap fst (zip x x) ≡ x

--
-- /Note/, the following doesn't hold:
--
